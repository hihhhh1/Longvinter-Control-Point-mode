local function show_killer_card(victim, killer)
    local k_name = "SUICIDE / ENVIRONMENT"
    local weapon_name = "-"
    local has_killer_player = false

    if killer ~= -1 and killer ~= victim then
        local k_info = game.player(killer)
        k_name = (k_info and k_info.name) or ("Player " .. tostring(killer))
        local k_data = players_data[killer]
        local k_weapon_key = (k_data and k_data.last_weapon) or "semipistol"
        weapon_name = (WEAPONS[k_weapon_key] and WEAPONS[k_weapon_key].name) or "Semi-Auto Pistol"
        has_killer_player = true
    end

    local html = string.format([[
<div class="killer-card">
    <img id="killer_avatar" class="killer-avatar"/>
    <div class="killer-details">
        <div class="killer-label">KILLED BY</div>
        <div id="killer_name" class="killer-name">%s</div>
        <div id="killer_weapon" class="killer-weapon">%s</div>
    </div>
</div>
]], k_name, weapon_name)

    ui.show(victim, "killer_info", html, KILLER_CSS, false)

    if has_killer_player then
        ui.set_avatar(victim, "killer_info", "killer_avatar", killer)
    end
end

local function is_in_hill(pos)
    if not pos then return false end
    return pos.x >= HILL_AREA.min_x and pos.x <= HILL_AREA.max_x and
           pos.y >= HILL_AREA.min_y and pos.y <= HILL_AREA.max_y and
           pos.z >= HILL_AREA.min_z and pos.z <= HILL_AREA.max_z
end

local function get_random_spawn(team)
    local list = SPAWN_POINTS[team] or SPAWN_POINTS.red
    return list[math.random(1, #list)]
end

local function format_time(seconds)
    local m = math.floor(seconds / 60)
    local s = seconds % 60
    return string.format("%02d:%02d", m, s)
end

local function update_all_huds()
    local red_str = string.format("RED: %d", state.red_score)
    local blue_str = string.format("BLUE: %d", state.blue_score)
    local base_time = format_time(state.time_left)

    local status_changed = (state.status ~= state.last_hud_status)

    for _, pid in ipairs(game.players()) do
        if status_changed then
            ui.show(pid, "koth_hud", get_hud_html(state.status), HUD_CSS, false)
        end

        ui.set_text(pid, "koth_hud", "red_score", red_str)
        ui.set_text(pid, "koth_hud", "blue_score", blue_str)
        ui.set_text(pid, "koth_hud", "center_box", base_time)
    end

    if status_changed then
        state.last_hud_status = state.status
    end
end

local function give_loadout(player_id, weapon_key)
    local w = WEAPONS[weapon_key] or WEAPONS.semipistol

    if players_data[player_id] then
        players_data[player_id].last_weapon = weapon_key
    end

    game.clear_inventory(player_id, false)
    game.give_item(player_id, w.id, 1)
    game.equip_item(player_id, 1)

    if w.ammo > 0 then
        game.set_ammo(player_id, w.ammo, false)
    end
end

local function assign_team(player_id)
    local red_count = 0
    local blue_count = 0

    for id, p in pairs(players_data) do
        if id ~= player_id then
            if p.team == "red" then red_count = red_count + 1
            elseif p.team == "blue" then blue_count = blue_count + 1 end
        end
    end

    local team = (red_count <= blue_count) and "red" or "blue"
    players_data[player_id] = players_data[player_id] or { kills = 0, deaths = 0, last_weapon = "semipistol", is_dead = false }
    players_data[player_id].team = team
    players_data[player_id].is_dead = false
    game.set_team(player_id, team)

    if team == "red" then
        game.set_outline(player_id, 1.0, 0.2, 0.2)
    else
        game.set_outline(player_id, 0.2, 0.5, 1.0)
    end

    return team
end

local function open_scoreboard(player_id)
    ui.open(player_id, "scoreboard", true)

    ui.set_text(player_id, "scoreboard", "sb_red_title", string.format("RED TEAM (%d)", state.red_score))
    ui.set_text(player_id, "scoreboard", "sb_blue_title", string.format("BLUE TEAM (%d)", state.blue_score))

    local red_idx = 1
    local blue_idx = 1

    for _, pid in ipairs(game.players()) do
        local data = players_data[pid] or { team = "red", kills = 0, deaths = 0 }
        local p_info = game.player(pid)
        local p_name = (p_info and p_info.name) or ("Player " .. tostring(pid))
        local name_str = string.format("%s (%dK / %dD)", p_name, data.kills, data.deaths)

        if data.team == "red" and red_idx <= 6 then
            ui.set_text(player_id, "scoreboard", "sb_r_txt_" .. tostring(red_idx), name_str)
            ui.set_avatar(player_id, "scoreboard", "sb_r_img_" .. tostring(red_idx), pid)
            red_idx = red_idx + 1
        elseif data.team == "blue" and blue_idx <= 6 then
            ui.set_text(player_id, "scoreboard", "sb_b_txt_" .. tostring(blue_idx), name_str)
            ui.set_avatar(player_id, "scoreboard", "sb_b_img_" .. tostring(blue_idx), pid)
            blue_idx = blue_idx + 1
        end
    end

    for i = red_idx, 6 do
        ui.set_text(player_id, "scoreboard", "sb_r_txt_" .. tostring(i), "-")
    end
    for i = blue_idx, 6 do
        ui.set_text(player_id, "scoreboard", "sb_b_txt_" .. tostring(i), "-")
    end
end

local function update_hill()
    local red_count = 0
    local blue_count = 0

    for _, pid in ipairs(game.players()) do
        local data = players_data[pid]
        if data and not data.is_dead then
            local pos = game.position(pid)
            if pos and is_in_hill(pos) then
                if data.team == "red" then red_count = red_count + 1
                elseif data.team == "blue" then blue_count = blue_count + 1 end
            end
        end
    end

    state.hill_occupants.red = red_count
    state.hill_occupants.blue = blue_count

    if red_count > 0 and blue_count > 0 then
        state.status = "yellow"
    elseif red_count > 0 and blue_count == 0 then
        state.status = "red"
        state.red_score = state.red_score + 1
    elseif blue_count > 0 and red_count == 0 then
        state.status = "blue"
        state.blue_score = state.blue_score + 1
    else
        state.status = "gray"
    end
end

local function restart_match()
    state.red_score = 0
    state.blue_score = 0
    state.time_left = CONFIG.ROUND_TIME
    state.game_over = false
    state.status = "gray"
    state.last_hud_status = ""

    for _, pid in ipairs(game.players()) do
        ui.close(pid, "scoreboard")

        if players_data[pid] then
            players_data[pid].kills = 0
            players_data[pid].deaths = 0
            players_data[pid].is_dead = false
        end

        local team = (players_data[pid] and players_data[pid].team) or "red"
        local spawn_guid = get_random_spawn(team)
        game.teleport(pid, spawn_guid)
        game.after(0.1, "force_tp_" .. tostring(pid) .. "_" .. spawn_guid)

        local weapon = (players_data[pid] and players_data[pid].last_weapon) or "semipistol"
        give_loadout(pid, weapon)
        game.set_invulnerable(pid, true)
        game.after(CONFIG.RESPAWN_DELAY, "remove_invuln_" .. tostring(pid))
    end

    game.set_phase("LIVE", CONFIG.ROUND_TIME)
    update_all_huds()
    game.after(1, "main_tick")
end

local function check_game_over()
    if state.game_over then return true end

    local winner = nil
    if state.red_score >= CONFIG.TARGET_SCORE then
        winner = "RED TEAM WINS!"
    elseif state.blue_score >= CONFIG.TARGET_SCORE then
        winner = "BLUE TEAM WINS!"
    elseif state.time_left <= 0 then
        if state.red_score > state.blue_score then
            winner = "TIME OVER - RED TEAM WINS!"
        elseif state.blue_score > state.red_score then
            winner = "TIME OVER - BLUE TEAM WINS!"
        else
            winner = "TIME OVER - DRAW!"
        end
    end

    if winner then
        state.game_over = true
        game.set_phase(winner, 0)

        for _, pid in ipairs(game.players()) do
            open_scoreboard(pid)
        end

        game.after(5, "end_game_delay")
        return true
    end
    return false
end

local function game_tick()
    if state.game_over then return end

    if state.round_started then
        state.time_left = math.max(0, state.time_left - 1)
        update_hill()
    end
    update_all_huds()

    if not check_game_over() then
        game.after(1, "main_tick")
    end
end

function on_start()
    game.enable_rounds(true)
    game.set_friendly_fire(false)
    game.after(1, "main_tick")
end

function on_timer(label)
    if label == "main_tick" then
        game_tick()
    elseif label == "end_game_delay" then
        restart_match()
    elseif string.sub(label, 1, 8) == "respawn_" then
        local pid_str = string.sub(label, 9)
        local pid = tonumber(pid_str)
        if pid and players_data[pid] then
            local team = players_data[pid].team or "red"
            local spawn_guid = get_random_spawn(team)
            game.spawn_player(pid, spawn_guid)
            game.after(0.1, "force_tp_" .. tostring(pid) .. "_" .. spawn_guid)
        end
    elseif string.sub(label, 1, 9) == "force_tp_" then
        local rest = string.sub(label, 10)
        local underscore_pos = string.find(rest, "_")
        if underscore_pos then
            local pid = tonumber(string.sub(rest, 1, underscore_pos - 1))
            local spawn_guid = string.sub(rest, underscore_pos + 1)
            if pid and spawn_guid and spawn_guid ~= "" then
                game.teleport(pid, spawn_guid)
            end
        end
    elseif string.sub(label, 1, 14) == "remove_invuln_" then
        local pid_str = string.sub(label, 15)
        local pid = tonumber(pid_str)
        if pid then
            game.set_invulnerable(pid, false)
        end
    end
end

function on_player_event(event, player_id)
    if event == "joined" then
        players_data[player_id] = { kills = 0, deaths = 0, last_weapon = "semipistol", is_dead = false }
        local team = assign_team(player_id)
        local spawn_guid = get_random_spawn(team)
        
        ui.show(player_id, "koth_hud", get_hud_html(state.status), HUD_CSS, false)
        game.set_respawn(player_id, true, spawn_guid, 0)
        game.spawn_player(player_id, spawn_guid)
        game.after(0.1, "force_tp_" .. tostring(player_id) .. "_" .. spawn_guid)

        if not state.round_started then
            state.round_started = true
            game.set_phase("LIVE", CONFIG.ROUND_TIME)
        end

    elseif event == "spawned" then
        local data = players_data[player_id]
        if not data then
            players_data[player_id] = { kills = 0, deaths = 0, last_weapon = "semipistol", is_dead = false }
            data = players_data[player_id]
        end
        data.is_dead = false

        local team = data.team or assign_team(player_id)
        local spawn_guid = get_random_spawn(team)

        game.teleport(player_id, spawn_guid)
        game.after(0.1, "force_tp_" .. tostring(player_id) .. "_" .. spawn_guid)

        game.set_team(player_id, team)
        if team == "red" then
            game.set_outline(player_id, 1.0, 0.2, 0.2)
        else
            game.set_outline(player_id, 0.2, 0.5, 1.0)
        end

        local weapon = data.last_weapon or "semipistol"
        give_loadout(player_id, weapon)
        game.set_invulnerable(player_id, true)
        game.after(CONFIG.RESPAWN_DELAY, "remove_invuln_" .. tostring(player_id))

        ui.close(player_id, "killer_info")

        local next_spawn = get_random_spawn(team)
        game.set_respawn(player_id, true, next_spawn, CONFIG.RESPAWN_DELAY)

    elseif event == "left" then
        players_data[player_id] = nil
    end
end

function on_kill(victim, killer, time)
    if players_data[victim] then
        players_data[victim].deaths = players_data[victim].deaths + 1
        players_data[victim].is_dead = true

        show_killer_card(victim, killer)

        game.after(CONFIG.RESPAWN_DELAY, "respawn_" .. tostring(victim))
    end

    if killer ~= -1 and players_data[killer] then
        players_data[killer].kills = players_data[killer].kills + 1
        game.add_score(killer, 10)
    end
end

function on_chat(player_id, message, channel)
    local msg = string.lower(message)
    msg = string.match(msg, "^%s*(.-)%s*$") or msg

    if msg == "!tab" or msg == "tab" or msg == "!score" or msg == "score" or msg == "/tab" or msg == "/score" then
        open_scoreboard(player_id)
    elseif msg == "!buy" or msg == "buy" or msg == "/buy" or msg == "!shop" or msg == "shop" or msg == "!armory" or msg == "armory" or msg == "/shop" or msg == "/armory" then
        ui.open(player_id, "weapon_shop", true)
    end
end

function on_ui_action(player_id, panel_name, action)
    if panel_name == "koth_hud" then
        if action == "open_armory" then
            ui.open(player_id, "weapon_shop", true)
        elseif action == "open_scoreboard" then
            open_scoreboard(player_id)
        end
    elseif panel_name == "weapon_shop" then
        if action == "select_shotgun" then
            give_loadout(player_id, "shotgun")
            ui.close(player_id, "weapon_shop")
        elseif action == "select_smg" then
            give_loadout(player_id, "smg")
            ui.close(player_id, "weapon_shop")
        elseif action == "select_lmg" then
            give_loadout(player_id, "lmg")
            ui.close(player_id, "weapon_shop")
        elseif action == "select_autopistol" then
            give_loadout(player_id, "autopistol")
            ui.close(player_id, "weapon_shop")
        elseif action == "select_sniper" then
            give_loadout(player_id, "sniper")
            ui.close(player_id, "weapon_shop")
        elseif action == "select_semipistol" then
            give_loadout(player_id, "semipistol")
            ui.close(player_id, "weapon_shop")
        elseif action == "select_machete" then
            give_loadout(player_id, "machete")
            ui.close(player_id, "weapon_shop")
        elseif action == "close_shop" then
            ui.close(player_id, "weapon_shop")
        end
    elseif panel_name == "scoreboard" then
        if action == "close_scoreboard" then
            ui.close(player_id, "scoreboard")
        end
    end
end
